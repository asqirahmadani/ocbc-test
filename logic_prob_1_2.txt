func main() {
	log.Print(ValidParentheses())
}

func ValidParentheses() bool {
	data := "{([)]}"

	stack := []rune{}

	for _, val:= range data {
		if val == '(' {
			stack = append(stack, ')')
		} else if val == '{' {
			stack = append(stack, '}')
		} else if val == '[' {
			stack = append(stack, ']')
		} else {
			if len(stack) == 0 || stack[len(stack) - 1] != val {
				return false
			}
			stack = stack[:len(stack)-1]
		}
	}

	return len(stack) == 0
}
