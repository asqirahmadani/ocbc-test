func isPrimary(grocery item, quantity int) float64 {
	var price float64

	if grocery.sale != nil {
		sisa := quantity % grocery.sale.unit
		price += float64(sisa) * grocery.price

		price += float64(quantity) * grocery.sale.price / float64(grocery.sale.unit)
	} else {
		price = grocery.price * float64(quantity)
	}

	return price
}
